cmake_minimum_required(VERSION 3.20)
project(TensorForge VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(LLVM REQUIRED CONFIG)
message(STATUS "LLVM ${LLVM_PACKAGE_VERSION} at ${LLVM_DIR}")
if(NOT LLVM_VERSION_MAJOR EQUAL 23)
  message(FATAL_ERROR "TensorForge currently targets LLVM 23; other major versions have not been validated")
endif()
option(TENSORFORGE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer for host code" OFF)
add_library(tf_options INTERFACE)
target_compile_options(tf_options INTERFACE -Wall -Wextra -Wpedantic -Wconversion -Wno-missing-field-initializers -ffp-contract=off)
if(TENSORFORGE_SANITIZERS)
  target_compile_options(tf_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(tf_options INTERFACE -fsanitize=address,undefined)
endif()
add_library(tf_core src/frontend.cpp src/ir.cpp src/interpreter.cpp)
target_include_directories(tf_core PUBLIC include)
target_link_libraries(tf_core PRIVATE tf_options)
add_library(tf_llvm src/codegen.cpp)
target_include_directories(tf_llvm SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
separate_arguments(TF_LLVM_DEFINITIONS NATIVE_COMMAND "${LLVM_DEFINITIONS}")
target_compile_options(tf_llvm PRIVATE ${TF_LLVM_DEFINITIONS})
llvm_map_components_to_libnames(TF_LLVM_LIBS core orcjit native passes)
target_link_libraries(tf_llvm PUBLIC tf_core PRIVATE tf_options ${TF_LLVM_LIBS})
add_executable(tensorforge src/main.cpp benchmarks/benchmark.cpp)
target_link_libraries(tensorforge PRIVATE tf_llvm tf_options)
target_compile_definitions(tensorforge PRIVATE TF_SANITIZERS="$<BOOL:${TENSORFORGE_SANITIZERS}>" TF_BUILD_TYPE="$<CONFIG>" TF_COMPILER="${CMAKE_CXX_COMPILER_ID}-${CMAKE_CXX_COMPILER_VERSION}")
include(CTest)
if(BUILD_TESTING)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(tf_tests tests/tests.cpp)
  target_link_libraries(tf_tests PRIVATE tf_llvm tf_options)
  foreach(group lexer parser semantic ir passes codegen differential resources runtime)
    add_test(NAME ${group} COMMAND tf_tests ${group})
  endforeach()
  foreach(example relu_chain scalar_folding dead_code scalar_only scalar_input identity shared_dag matrix_broadcast)
    add_test(NAME example_${example} COMMAND tensorforge run ${CMAKE_CURRENT_SOURCE_DIR}/examples/${example}.tf --opt --verify)
  endforeach()
  add_test(NAME cli_negative COMMAND ${CMAKE_COMMAND} -DCLI=$<TARGET_FILE:tensorforge> -DSOURCES=${CMAKE_CURRENT_SOURCE_DIR} -P ${CMAKE_CURRENT_SOURCE_DIR}/cmake/negative-tests.cmake)
  foreach(group options benchmark)
    add_test(NAME cli_${group} COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/cli_tests.py $<TARGET_FILE:tensorforge> ${CMAKE_CURRENT_SOURCE_DIR} ${group} $<CONFIG>)
  endforeach()
endif()
install(TARGETS tensorforge RUNTIME DESTINATION bin)
